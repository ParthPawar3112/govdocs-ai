# Phase 1 - Project Setup
# These are the ONLY packages needed to run the health-check backend.
# Deliberately excludes JWT/auth libraries, EasyOCR, and the Gemini SDK --
# those are added to this file only when we build the module that needs
# them (Login, OCR, AI Metadata Extraction). This keeps today's install
# fast (seconds, not minutes) and avoids pulling in heavy packages
# (e.g. EasyOCR depends on PyTorch, a multi-GB download) before they're
# actually used.

fastapi==0.115.6
uvicorn[standard]==0.32.1
# Bumped from 2.10.3 in Phase 6: google-genai requires pydantic>=2.12.5.
# Verified compatible with fastapi==0.115.6 and every existing schema/config
# in this project - see Phase 6 testing notes.
pydantic==2.12.5
pydantic-settings==2.14.2
python-dotenv==1.0.1
sqlalchemy==2.0.36
passlib[bcrypt]==1.7.4
# Passlib 1.7.4 accesses bcrypt's legacy version metadata; 4.0.1 is its
# latest compatible bcrypt release and avoids a runtime startup traceback.
bcrypt==4.0.1
python-jose[cryptography]==3.3.0
python-multipart==0.0.20

# --- OCR (Phase 5) ---
# Default engine: pytesseract + Tesseract binary. Chosen over EasyOCR by
# default because it has no PyTorch dependency, installs in seconds, and is
# reliable on Windows/newer Python where EasyOCR+torch wheels can lag behind.
# Windows also needs the Tesseract *binary* installed separately - see
# README "OCR Engine" section. To use EasyOCR instead, set OCR_ENGINE=easyocr
# in .env and separately `pip install easyocr` (not included here - it pulls
# in PyTorch, a multi-GB install).
pymupdf==1.28.0
pytesseract==0.3.13
Pillow==12.3.0

# --- AI Metadata Extraction (Phase 6) ---
# Official Google Gen AI SDK. Lightweight, pure-Python, no heavy transitive
# deps (unlike EasyOCR's torch) - installs in seconds.
google-genai==2.16.0

# --- Smart Search (Phase 7) ---
# Fuzzy-match fallback for typos/rewording - only invoked when the primary
# SQL search finds zero results (see search_service.py). Compiled but ships
# prebuilt wheels for all common platforms/Python versions - fast install.
rapidfuzz==3.14.5

# --- Approval Workflow & Reporting (Phase 8) ---
# Pure-Python PDF generation for the Summary Report export - no native
# binary dependency, same Windows-friendliness reasoning as pymupdf/pytesseract
# above. reportlab was considered but is heavier for what's a one-page report.
fpdf2==2.8.2
